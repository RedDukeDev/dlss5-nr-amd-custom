# Which translated kernels a native one stands in for.
#
# Point the proxy at this file with NVCUDA_PROXY_NATIVE. One kernel per line:
#
#   <ptx kernel name>  <code object>  block=x,y,z  args=<bytes>  [entry=<symbol>]
#
# block= and args= are not decoration: they are the gate. A build of DLSS whose
# kernel has a different block shape or a different parameter size is a kernel
# this native one was not written for, and the line below will refuse to run and
# leave the translation in place rather than produce a wrong image.
#
# Measured on 310.8.0.0, at 2560x1440, from the proxy's own launch log. The
# resolution is not a guess: 2560/16 = 160 and 1440/16 = 90 match the (160, 92)
# tile grid the kernels are handed in their parameter.
#
#   kernel                                             ms    %   block    grid
#   swin_8h_256_8_chained_fp8                       42.06  11.1  32x8x1   20x12x1
#   pre_block_swin_1h_32_1_ds_fp8                   30.56   8.0  32x1x1   320x184x1
#   swin_4h_128_4_chained_fp8                       30.29   8.0  32x4x1   40x23x1
#   post_block_swin_1h_32_fp8                       28.70   7.5  32x1x1   321x185x1
#   swin_1h_32_1_chained_fp8                        28.20   7.4  32x1x1   161x92x1
#
# The 88-byte parameter is passed by value, not by pointer (the PTX declares
# .param .align 8 .b8 <name>_param_0[88]). Its layout, cross-checked between
# what the kernel loads and what a real launch carried:
#
#   +0x00  const void*   input, into the working pool
#   +0x08  const void*   second input, same pool
#   +0x10  void*         output, into the weight/persistent pool
#   +0x18  --            never read
#   +0x20  int2          (160, 92) tiles, constant for a resolution
#   +0x28  int2          window shift, alternates between launches (0,0)/(-4,-4)
#   +0x30  const void*   third buffer
#   +0x38  int2          (0, 0)
#   +0x40  const void*   fourth buffer
#   +0x48  --            never read
#   +0x50  --            never read
#
# The same block again with eight heads and 256 channels, on eight warps. The
# split is the 4H one; what differs is the memory: eight warps and 256 channels
# do not fit with f16 buffers, so what crosses shared memory stays e4m3 and is
# widened where it is used -- sixteen bytes read in one go are one operand, one
# LDS read instead of two, and the values are the same to the bit. Checked on
# two captured launches (21x12 at shift (-4,-4) and 20x12 at (0,-4)) against the
# translation run here: every output byte and every ready flag identical.
cc_tinlayout_fused_swin_8h_256_8_chained_fp8  swin_8h.hsaco  block=32,8,1  args=88  entry=cc_tinlayout_fused_swin_8h_256_8_chained_fp8_native
